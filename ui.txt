    function colorRgba(hex, alpha) {
        const value = hex.replace('#','');
        return 'rgba(' + parseInt(value.slice(0,2),16) + ',' +

    function blendColor(hex, target, amount) {
        const a = hex.replace('#','');
        const b = target.replace('#','');
        const mix = (start, end) => Math.round(parseInt(start,16) * (1-amount) + parseInt(end,16) * amount)
            .toString(16).padStart(2,'0');
        return ('#' + mix(a.slice(0,2),b.slice(0,2)) + mix(a.slice(2,4),b.slice(2,4)) +
            mix(a.slice(4,6),b.slice(4,6))).toUpperCase();
    }

        const g = parseInt(value.slice(2,4),16) / 255;
        const b = parseInt(value.slice(4,6),16) / 255;
        const max = Math.max(r,g,b), min = Math.min(r,g,b), delta = max-min;
        let hue = 0;
        if (delta) {
            if (max === r) hue = 60 * (((g-b)/delta) % 6);
            else if (max === g) hue = 60 * ((b-r)/delta + 2);
            else hue = 60 * ((r-g)/delta + 4);
        }
        if (hue < 0) hue += 360;
        return Math.round((220 + hue - 110 + 360) % 360);
    }

    function applyAreaIconTintDirectly() {
        const color = activeColors.icon;
        const filterValue = 'hue-rotate(' + hueFilterFor(color) + 'deg) saturate(4)';
        document.querySelectorAll(attentionIconSelector).forEach((element) => {
            const svg = element.matches('svg') ? element : element.closest('svg');
            if (svg) {
                // Vector and font icons can use the exact same solid color as the alert dot.
                svg.style.setProperty('filter','none','important');
                svg.style.setProperty('color',color,'important');
                svg.querySelectorAll('path,circle,ellipse,line,polygon,polyline,rect').forEach((shape) => {
                    const fillAttribute = shape.getAttribute('fill');
                    const fillStyle = shape.style.fill;
                    const strokeAttribute = shape.getAttribute('stroke');
                    const strokeStyle = shape.style.stroke;
                    const computed = window.getComputedStyle(shape);
                    const hasFill = (fillAttribute && fillAttribute.toLowerCase() !== 'none') ||
                        (fillStyle && fillStyle !== 'none') ||
                        (!strokeAttribute && !strokeStyle && computed.fill !== 'none');
                    const hasStroke = (strokeAttribute && strokeAttribute.toLowerCase() !== 'none') ||
                        (strokeStyle && strokeStyle !== 'none') || computed.stroke !== 'none';
                    if (hasFill) shape.style.setProperty('fill',color,'important');
                    if (hasStroke) shape.style.setProperty('stroke',color,'important');
                });
            } else {
                const computed = window.getComputedStyle(element);
                const imageBased = element.matches('img,canvas') || computed.backgroundImage !== 'none';
                if (imageBased) {
                    // Keep the raster fallback, but apply it only to the icon itself.
                    element.style.setProperty('filter',filterValue,'important');
                } else {
                    // Icon fonts and pseudo-element glyphs inherit this exact color.
                    element.style.setProperty('filter','none','important');
                    element.style.setProperty('color',color,'important');
                }
            }
        });
    }

        const hslToHex = (hue,saturation,lightness) => {
            const chroma = (1-Math.abs(2*lightness-1))*saturation;
            const segment = hue/60;
            const second = chroma*(1-Math.abs(segment%2-1));
            let rgb;
            if (segment < 1) rgb = [chroma,second,0];
            else if (segment < 2) rgb = [second,chroma,0];
            else if (segment < 3) rgb = [0,chroma,second];
            else if (segment < 4) rgb = [0,second,chroma];
            else if (segment < 5) rgb = [second,0,chroma];
            else rgb = [chroma,0,second];
            const match = lightness-chroma/2;
            return '#'+rgb.map(channel => Math.round((channel+match)*255).toString(16).padStart(2,'0')).join('').toUpperCase();
        };

    function normalizePath(path) {
        const normalized = path.toLowerCase().replace(/\/+$/, '');
        return normalized || '/';
    }

    function getRouteKey(value) {
        let url;

        try {
            url = new URL(value, window.location.href);
        } catch {
            return null;
        }

        const path = normalizePath(url.pathname);

        if (path === '/page.php') {
            const sid = (url.searchParams.get('sid') || '').toLowerCase();
            return `${path}?sid=${sid}`;
        }

        if (path === '/companies.php') {
            const step = (url.searchParams.get('step') || '').toLowerCase();
            const type = (url.searchParams.get('type') || '').toLowerCase();
            return `${path}?step=${step}&type=${type}`;
        }

        return path;
    }

    function getCurrentAreaLabel() {
        const currentUrl = new URL(window.location.href);
        const currentPath = normalizePath(currentUrl.pathname);

        if (currentPath === '/page.php') {
            const sid = (currentUrl.searchParams.get('sid') || '').toLowerCase();
            return pageSidLabels.get(sid) || null;
        }

        const currentRouteKey = getRouteKey(currentUrl.href);

        return routeLabels.get(currentRouteKey) ||
            routeLabels.get(currentPath) ||
            null;
    }

    function getButtonLink(button) {
        if (button.matches('a')) {
            return button;
        }

        return button.closest('a') || button.querySelector('a');
    }

    function markButtonActive(button) {
        button.classList.add('keza-active-area-button');

        const pill = button.matches('.pill')
            ? button
            : button.closest('.pill') || button.querySelector('.pill');

        if (pill) {
            pill.classList.add('keza-active-area-button');
        }

        const link = getButtonLink(button);

        if (link) {
            link.classList.add('keza-active-area-button');
        }
    }

    function updateActiveAreaButton() {
        document.querySelectorAll('.keza-active-area-button').forEach((element) => {
            element.classList.remove('keza-active-area-button');
        });

        const currentRouteKey = getRouteKey(window.location.href);
        const currentLabel = getCurrentAreaLabel();
        const buttons = document.querySelectorAll(areaButtonSelector);

        buttons.forEach((button) => {
            let matchesCurrentRoute = false;
            const link = getButtonLink(button);

            if (link) {
                const href = link.getAttribute('href');

                if (
                    href &&
                    !href.startsWith('#') &&
                    !href.toLowerCase().startsWith('javascript:')
                ) {
                    const destinationRouteKey = getRouteKey(
                        new URL(href, window.location.href).href
                    );

                    matchesCurrentRoute =
                        destinationRouteKey !== null &&
                        destinationRouteKey === currentRouteKey;
                }
            }

            const buttonText = (button.innerText || button.textContent || '')
                .replace(/\s+/g, ' ')
                .trim()
                .toLowerCase();

            const matchesCurrentLabel =
                currentLabel &&
                buttonText.includes(currentLabel.toLowerCase());

            if (matchesCurrentRoute || matchesCurrentLabel) {
                markButtonActive(button);
            }
        });
    }

    function parseCssColor(value) {
        if (!value || value === 'transparent') {
            return null;
        }

        const numbers = value.match(/[\d.]+/g);

        if (!numbers || numbers.length < 3) {
            return null;
        }

        return {
            red: Number(numbers[0]),
            green: Number(numbers[1]),
            blue: Number(numbers[2]),
            alpha: numbers.length >= 4 ? Number(numbers[3]) : 1
        };
    }

    function getNeutralGrayValue(color) {
        if (!color || color.alpha < 0.15) {
            return null;
        }

        const max = Math.max(color.red, color.green, color.blue);
        const min = Math.min(color.red, color.green, color.blue);
        const brightness = (color.red + color.green + color.blue) / 3;

        if (max - min > 16 || brightness < 35 || brightness > 155) {
            return null;
        }

        return brightness;
    }

    function getGraySurfaceBrightness(computedStyle) {
        const directColor = parseCssColor(computedStyle.backgroundColor);
        const directBrightness = getNeutralGrayValue(directColor);

        if (directBrightness !== null) {
            return directBrightness;
        }

        const gradientColors =
            computedStyle.backgroundImage.match(/rgba?\([^)]*\)/gi) || [];

        for (const value of gradientColors) {
            const brightness = getNeutralGrayValue(parseCssColor(value));

            if (brightness !== null) {
                return brightness;
            }
        }

        return null;
    }

    function getFactionSurfaceBrightness(computedStyle) {
        const colors = [parseCssColor(computedStyle.backgroundColor)];
        const gradients = computedStyle.backgroundImage.match(/rgba?\([^)]*\)/gi) || [];
        gradients.forEach((value) => colors.push(parseCssColor(value)));

        for (const color of colors) {

            const min = Math.min(color.red, color.green, color.blue);
            const brightness = (color.red + color.green + color.blue) / 3;
            if (max - min <= 20 && brightness <= 155) {
                return brightness < 35 ? 35 : brightness;
            }
        }
        return null;
    }

    function markOriginalAppearanceRegions(root) {
        root.querySelectorAll('.' + preserveNativeClass).forEach((element) => {
            element.classList.remove(preserveNativeClass);
        });

        const normalizeText = (value) =>
            value.replace(/\s+/g, ' ').trim().toLowerCase();

        const markSubtree = (element) => {
            if (!element || element === root) {
                return;
            }

            element.classList.add(preserveNativeClass);
            element.querySelectorAll('*').forEach((child) => {
                child.classList.add(preserveNativeClass);
            });
        };

        const profileStatLabels = new Set([
            'level',
            'rank',
            'age',
            'playtime',
            'play time'
        ]);
        const userProfileRoot = root.querySelector('.user-profile');

        if (userProfileRoot) {
            userProfileRoot.querySelectorAll('.box-info.age').forEach((element) => {
                markSubtree(element);
            });
        }

        const tbsHeaderLabels = new Set(['tbs', '% you', 'ff', 'source', 'date']);
        const foundTbsHeaders = new Set();
        let firstTbsHeader = null;

        const textWalker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
        let textNode;

        while ((textNode = textWalker.nextNode())) {
            const label = normalizeText(textNode.nodeValue);

            if (
                profileStatLabels.has(label) &&
                textNode.parentElement.closest('.user-profile')
            ) {
                let current = textNode.parentElement;

                for (let depth = 0; current && current !== root && depth < 7; depth++) {
                    current.classList.add(preserveNativeClass);

                    const rect = current.getBoundingClientRect();

                    if (
                        depth > 0 &&
                        rect.width >= 70 &&
                        rect.width <= 340 &&
                        rect.height >= 35 &&
                        rect.height <= 240
                    ) {
                        markSubtree(current);
                        break;
                    }

                    if (rect.width > 450 || rect.height > 260) {
                        break;
                    }

                    current = current.parentElement;
                }
            }

            if (tbsHeaderLabels.has(label)) {
                foundTbsHeaders.add(label);
                firstTbsHeader ||= textNode.parentElement;
            }
        }

        const hasTbsHeaders = (element) => {
            const text = normalizeText(element.textContent || '');
            return text.includes('tbs') &&
                text.includes('% you') &&
                text.includes('ff') &&
                text.includes('source') &&
                text.includes('date');
        };

        const tableSelectors = [
            'table',
            '[role="table"]',
            '[role="grid"]',
            '[class*="table"]',
            '[class*="Table"]'
        ].join(', ');

        let tbsTableFound = false;

        for (const element of root.querySelectorAll(tableSelectors)) {
            if (hasTbsHeaders(element)) {
                markSubtree(element);
                tbsTableFound = true;
                break;
            }
        }

        if (
            !tbsTableFound &&
            firstTbsHeader &&
            ['tbs', '% you', 'ff', 'source', 'date']
                .every((label) => foundTbsHeaders.has(label))
        ) {
            let current = firstTbsHeader;

            for (let depth = 0; current && current !== root && depth < 9; depth++) {
                if (hasTbsHeaders(current)) {
                    const rect = current.getBoundingClientRect();

                    if (rect.width <= 700 && rect.height <= 360) {
                        markSubtree(current);
                        break;
                    }
                }

                current = current.parentElement;
            }
        }
    }

    function recolorGrayPageSurfaces() {
        const now = Date.now();

        if (now - lastPlumSurfaceScan < 600) {
            return;
        }

        lastPlumSurfaceScan = now;

        const root = document.querySelector('#mainContainer');

        if (!root) {
            return;
        }

        const excludedAreaSelector = [
            '#sidebarroot',
            '#sidebar',
            '.areasWrapper',
            '#header',
            '#header-old',
            '#topHeader',
            '[class*="header-old"]'
        ].join(', ');

        markOriginalAppearanceRegions(root);

        root.querySelectorAll('*').forEach((element) => {
            if (element.classList.contains(preserveNativeClass)) {
                element.classList.remove(
                    ...plumSurfaceClasses,
                    transparentLabelClass
                );
                return;
            }

            element.classList.remove(
                ...plumSurfaceClasses,
                transparentLabelClass
            );

            if (element.matches('[data-bsp-stats]')) {
                element.classList.remove(
                    'keza-faction-member-surface-dark',
                    'keza-faction-member-surface-medium',
                    'keza-faction-member-surface-light'
                );
                return;
            }

            if (
                element.matches(
                    'img, svg, path, video, canvas, input, textarea, select, button'
                )
            ) {
                return;
            }

            if (element.closest(excludedAreaSelector)) {
                return;
            }

            if (element.closest('span.divider')) {
                return;
            }

            const rect = element.getBoundingClientRect();

            if (rect.height < 16) {
                return;
            }

            const computedStyle = window.getComputedStyle(element);

            if (
                computedStyle.backgroundImage.includes('url(') &&
                !computedStyle.backgroundImage.includes('gradient')
            ) {
                return;
            }

            const brightness = getGraySurfaceBrightness(computedStyle);

            if (brightness === null) {
                return;
            }

            const isSmallTextLabel =
                element.matches('span, label, strong, b, em, small') &&
                rect.width <= 320 &&
                rect.height <= 42 &&
                (element.innerText || element.textContent || '').trim().length > 0;

            if (isSmallTextLabel) {
                element.classList.add(transparentLabelClass);
                return;
            }

            if (rect.width < 90) {
                return;
            }

            const viewportArea = window.innerWidth * window.innerHeight;
            const elementArea = rect.width * rect.height;

            if (viewportArea > 0 && elementArea > viewportArea * 0.42) {
                return;
            }

            element.classList.add(getPlumSurfaceClass(brightness));
        });
    }

    function normalizeBackgroundUrl(value) {
        try {
            const url = new URL(String(value || '').trim());
            return url.protocol === 'https:' || url.protocol === 'http:' ? url.href : null;
        } catch (_) {
            return null;
        }
    }

    function loadBackgroundPreference() {
        try {
            const saved = normalizeBackgroundUrl(localStorage.getItem(BACKGROUND_PREF_KEY));
            if (saved) return saved;
        } catch (_) {}
        return originalBackgroundUrl;
    }

    function applyBackgroundMedia(value,onResult) {
        const url = normalizeBackgroundUrl(value);
        if (!url || !document.body) {
            if (onResult) onResult(false);
            return;
        }

        const isVideo = /\.(mp4|webm|ogv|ogg)$/i.test(new URL(url).pathname);
        const media = document.createElement(isVideo ? 'video' : 'img');
        let finished = false;
        const previous = document.getElementById('local-video-background');
        media.id = 'keza-alpha-background-pending';
        media.style.cssText = 'position:fixed!important;inset:0!important;width:100vw!important;height:100vh!important;z-index:0!important;object-fit:cover!important;pointer-events:none!important;opacity:0!important;transition:opacity .2s linear!important;';

        const finish = (success) => {
            if (finished) return;
            finished = true;
            if (success) {
                if (previous && previous.isConnected) previous.remove();
                media.id = 'local-video-background';
                media.style.setProperty('opacity','1','important');
                refreshMotionPolicy();
            } else {
                media.remove();
            }
            if (onResult) onResult(success);
        };

        if (isVideo) {
            media.autoplay = true;
            media.muted = true;
            media.loop = true;
            media.playsInline = true;
            media.preload = 'auto';
            media.playbackRate = motionSettings.backgroundVideoSpeed;
            media.addEventListener('canplay',() => finish(true),{once:true});
            media.addEventListener('error',() => finish(false),{once:true});
        } else {
            media.alt = '';
            media.decoding = 'async';
            media.addEventListener('load',() => finish(true),{once:true});
            media.addEventListener('error',() => finish(false),{once:true});
        }

        media.src = url;
        document.body.prepend(media);
        if (isVideo && motionSettings.backgroundVideoEnabled && !motionIsReduced() && !document.hidden) {
            media.play().catch(() => {});
        }
    }

    function addBackground() {
        if (!document.body || document.getElementById('local-video-background')) {
            return;
        }

        ensureColorSettingsMenu();
        applyThemeColors();

        const loadingLayer = document.createElement('div');
        loadingLayer.id = 'torn-background-loading';
        document.body.prepend(loadingLayer);

        const themeOverlay = document.createElement('div');
        themeOverlay.id = 'keza-alpha-video-overlay';
        document.body.prepend(themeOverlay);

        const requestedBackground = loadBackgroundPreference();
        applyBackgroundMedia(requestedBackground,(success) => {
            if (success) {
                loadingLayer.remove();
            } else if (requestedBackground !== originalBackgroundUrl) {
                try { localStorage.removeItem(BACKGROUND_PREF_KEY); } catch (_) {}
                applyBackgroundMedia(originalBackgroundUrl,(restored) => {
                    if (restored) loadingLayer.remove();
                });
            } else {
                loadingLayer.remove();
                console.error('Background media failed to load.');
            }
        });

        updateSidebarStyles();

        const pageObserver = new MutationObserver(scheduleSidebarUpdate);
        pageObserver.observe(document.body, {
            childList: true,
            subtree: true
        });

        window.addEventListener('popstate', scheduleSidebarUpdate);
        window.addEventListener('hashchange', scheduleSidebarUpdate);
    }

    function initialize() {
        if (!document.body) {
            requestAnimationFrame(initialize);
            return;
        }

        watchColorSettingsMount();
        ensureColorSettingsMenu();
        applyThemeColors();
        addBackground();
    }

    initialize();
})();
